📐 Verified Power BI DAX Measures Library

Reconciled identically to SQL queries in sql/business_queries.sql

Total Revenue =

SUM(FactSales[line_net_revenue_usd])

// Expected Value: $497,018,086.85

Gross Margin % =

DIVIDE([Total Gross Profit], [Total Revenue], 0)

// Expected Value: 35.81%

Total Net Profit (Waterfall Formulation) =

[Total Gross Profit] - [Total Shipping Cost] - [Total Refunds] - ([Total Revenue] * 0.08)

// Expected Value: $115,363,353.25 (23.21%)

Avg Delivery Delay (Days) =

AVERAGE(FactShipping[delivery_delay_days])

// Expected Value: 2.39 days

Revenue YoY % =

VAR RevLY = CALCULATE([Total Revenue], SAMEPERIODLASTYEAR(DimDate[full_date]))
RETURN DIVIDE([Total Revenue] - RevLY, RevLY, 0)
